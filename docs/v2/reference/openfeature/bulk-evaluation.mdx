```bash cURL

  --url http://localhost:8080/ofrep/v1/evaluate/flags \
  --header 'Content-Type: application/json' \

  --data '{
  "context": {
    "flags": ["flagKey1", "flagKey2", "flagKey3"],
    "targetingKey": "targetingKey1"
  }
}'
```
